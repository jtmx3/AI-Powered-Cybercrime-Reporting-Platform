                        <div class="form-check mt-2">
                            <input id="formalReportConsent" type="checkbox" class="form-check-input" @bind="formalReportConsentConfirmed" disabled="@isSaving" />

@code {
    private enum ChatStage { Story, Incident, Timing, Urgency, Additional, Path, Review, Saved }
    private sealed record ChatMessage(bool IsUser, string Text);
    private static readonly (string Code, string Label)[] IncidentOptions =
    [
        ("Blackmail", "ابتزاز إلكتروني"), ("Fraud", "احتيال مالي"),
        ("AccountHacking", "اختراق حساب"), ("IdentityTheft", "انتحال شخصية"),
        ("Threat", "تهديد إلكتروني"), ("Other", "أخرى / غير متأكد")
    ];
    private static readonly (SeverityLevel Level, string Label)[] UrgencyOptions =
    [
        (SeverityLevel.Low, "منخفض — استفسار أو مشكلة غير مستعجلة"),
        (SeverityLevel.Medium, "متوسط — المشكلة قائمة وتحتاج متابعة"),
        (SeverityLevel.High, "مرتفع — ضرر أو تهديد مستمر"),
        (SeverityLevel.Critical, "حرج — خطر فوري يحتاج تدخلاً عاجلاً")
    ];
    private ChatStage stage = ChatStage.Story;
    private string story = string.Empty;
    private string incidentType = string.Empty;
    private string timing = string.Empty;
    private string additional = string.Empty;
    private string answerDraft = string.Empty;
    private SeverityLevel urgency = SeverityLevel.Medium;
    private CasePath selectedPath = CasePath.Assistance;
    private bool formalReportConsentConfirmed;
    private bool isSaving;
    private string errorMessage = string.Empty;
    private string successMessage = string.Empty;
    private string savedDetailsUrl = string.Empty;

    private string IncidentLabel => IncidentOptions.FirstOrDefault(x => x.Code == incidentType).Label ?? "غير محدد";
    private string UrgencyLabel => UrgencyOptions.FirstOrDefault(x => x.Level == urgency).Label ?? "غير محدد";
    private string PathLabel => selectedPath == CasePath.FormalReport ? "بلاغ رسمي" : "طلب مساعدة";
    private string RequestTitle
    {
        get
        {
            var title = $"{IncidentLabel}: {story}".Replace('\r', ' ').Replace('\n', ' ');
            return title[..Math.Min(title.Length, 150)];
        }
    }
    private string BaseDescription => $"{story}\n\nوقت الحادثة بحسب المستخدم: {timing}";
    private string RequestDescription => BaseDescription + (string.IsNullOrWhiteSpace(additional) ? string.Empty : $"\n\nتفاصيل إضافية بحسب المستخدم: {additional}");
    private int AdditionalLimit => Math.Max(0, 4000 - BaseDescription.Length - "\n\nتفاصيل إضافية بحسب المستخدم: ".Length);
    private int AnswerLimit => stage == ChatStage.Story ? 3000 : stage == ChatStage.Timing ? 300 : AdditionalLimit;
    private string ComposerLabel => stage == ChatStage.Story ? "اكتب ما حدث معك" : stage == ChatStage.Timing ? "متى بدأت المشكلة؟" : "هل تريد إضافة شيء؟";
    private string ComposerPlaceholder => stage == ChatStage.Story ? "احكِ بطريقتك: ماذا حدث؟ وعلى أي تطبيق أو موقع؟" : stage == ChatStage.Timing ? "مثلاً: أمس مساءً، أو منذ أسبوع، أو لا أتذكر" : "أي تفاصيل أخرى تريد تضمينها في الطلب";

    private List<ChatMessage> Conversation
    {
        get
        {
            var items = new List<ChatMessage> { new(false, "أهلاً بك. احكِ لي شو صار معك وعلى أي تطبيق أو موقع. سنرتّب المعلومات معاً، ولن يُرسل أي بلاغ قبل مراجعتك وموافقتك.") };
            if (stage > ChatStage.Story) { items.Add(new(true, story)); items.Add(new(false, "أي نوع من المشكلات هو الأقرب لما وصفته؟ يمكنك اختيار «أخرى / غير متأكد».")); }
            if (stage > ChatStage.Incident) { items.Add(new(true, IncidentLabel)); items.Add(new(false, "متى بدأت المشكلة؟ اذكر الوقت التقريبي، أو اكتب أنك لا تتذكر.")); }
            if (stage > ChatStage.Timing) { items.Add(new(true, timing)); items.Add(new(false, "كيف تصف استعجال المشكلة الآن؟ اختر الأقرب لوضعك.")); }
            if (stage > ChatStage.Urgency) { items.Add(new(true, UrgencyLabel)); items.Add(new(false, "هل هناك تفاصيل أخرى تريد إضافتها؟ مثل ما طلبه الطرف الآخر أو الضرر الذي وقع. يمكنك تجاوز هذه الخطوة.")); }
            if (stage > ChatStage.Additional) { items.Add(new(true, string.IsNullOrWhiteSpace(additional) ? "لا يوجد شيء آخر." : additional)); items.Add(new(false, "كيف تريد حفظ المعلومات؟ طلب المساعدة يبقى في حسابك، أما البلاغ الرسمي فيظهر للموظفين بعد موافقتك.")); }
            if (stage > ChatStage.Path) { items.Add(new(true, PathLabel)); items.Add(new(false, "هذا ملخص المعلومات التي قدمتها. راجعه قبل الحفظ، ويمكنك الرجوع لتعديل الإجابات.")); }
            if (stage == ChatStage.Saved) items.Add(new(false, successMessage));
            return items;
        }
    }

    private void SendAnswer()
    {
        if (isSaving || stage is not (ChatStage.Story or ChatStage.Timing or ChatStage.Additional)) return;
        errorMessage = string.Empty;
        var answer = answerDraft.Trim();
        if (answer.Length > AnswerLimit) { errorMessage = $"اختصر الإجابة إلى {AnswerLimit} حرفًا أو أقل."; return; }
        if (stage == ChatStage.Story && answer.Length < 10) { errorMessage = "احكِ لنا تفاصيل أكثر عن المشكلة (10 أحرف على الأقل)."; return; }
        if (stage == ChatStage.Timing && answer.Length == 0) { errorMessage = "اكتب الوقت التقريبي، أو اكتب «لا أتذكر»."; return; }
        if (stage == ChatStage.Story) story = answer;
        else if (stage == ChatStage.Timing) timing = answer;
        else additional = answer;
        Advance();
    }

    private void ChooseIncident(string code)
    {
        if (isSaving || stage != ChatStage.Incident || !IncidentOptions.Any(x => x.Code == code)) return;
        incidentType = code;
        Advance();
    }

    private void ChooseUrgency(SeverityLevel level)
    {
        if (isSaving || stage != ChatStage.Urgency || !Enum.IsDefined(level)) return;
        urgency = level;
        Advance();
    }

    private void SkipAdditional()
    {
        if (isSaving || stage != ChatStage.Additional) return;
        additional = string.Empty;
        Advance();
    }

    private void ChoosePath(CasePath path)
    {
        if (isSaving || stage != ChatStage.Path || !Enum.IsDefined(path)) return;
        selectedPath = path;
        formalReportConsentConfirmed = false;
        Advance();
    }

    private void Advance()
    {
        stage = (ChatStage)((int)stage + 1);
        errorMessage = string.Empty;
        LoadDraft();
    }

    private void GoBack()
    {
        if (isSaving || stage <= ChatStage.Story || stage >= ChatStage.Saved) return;
        stage = (ChatStage)((int)stage - 1);
        formalReportConsentConfirmed = false;
        errorMessage = string.Empty;
        LoadDraft();
    }

    private void LoadDraft() => answerDraft = stage == ChatStage.Story ? story : stage == ChatStage.Timing ? timing : stage == ChatStage.Additional ? additional : string.Empty;

    private void StartNewConversation()
    {
        if (isSaving || stage != ChatStage.Saved) return;
        stage = ChatStage.Story;
        story = incidentType = timing = additional = answerDraft = errorMessage = successMessage = savedDetailsUrl = string.Empty;
        selectedPath = CasePath.Assistance;
        urgency = SeverityLevel.Medium;
        formalReportConsentConfirmed = false;
    }

    private async Task SaveCase()
    {
        if (isSaving || stage != ChatStage.Review) return;
        errorMessage = string.Empty;
        if (!Enum.IsDefined(selectedPath) || !Enum.IsDefined(urgency) || !IncidentOptions.Any(x => x.Code == incidentType)
            || story.Trim().Length < 10 || story.Length > 3000 || string.IsNullOrWhiteSpace(timing) || timing.Length > 300
            || RequestDescription.Length > 4000)
        {
            errorMessage = "راجع إجاباتك: يجب اكتمال الوصف ونوع المشكلة والوقت، وألا يتجاوز الوصف الكامل 4000 حرف.";
            return;
        }
        if (selectedPath == CasePath.FormalReport && !formalReportConsentConfirmed)
        {
            errorMessage = "يجب الموافقة على تقديم البلاغ الرسمي قبل حفظه.";
            return;
        }
        var submission = new CyberCase
        {
            Title = RequestTitle, Description = RequestDescription, IncidentType = incidentType,
            SeverityLevel = urgency, Path = selectedPath
        };
        isSaving = true;
        try
        {
            var authenticationState = await AuthenticationStateProvider.GetAuthenticationStateAsync();
            var user = authenticationState.User;
            if (user.Identity?.IsAuthenticated != true || !user.IsInRole("USER")
                || !Guid.TryParse(user.FindFirstValue(ClaimTypes.NameIdentifier), out var reporterId) || reporterId == Guid.Empty)
            {
                errorMessage = "تعذر التعرف على حسابك. يرجى تسجيل الدخول مجددًا.";
                return;
            }
            var savedCase = await CaseService.CreateForReporterAsync(submission, reporterId);
            successMessage = savedCase.Path == CasePath.Assistance
                ? $"تم حفظ طلب المساعدة في حسابك. رقم الطلب: {savedCase.CaseNumber}"
                : $"تم تقديم البلاغ الرسمي للموظفين المختصين. رقم البلاغ: {savedCase.CaseNumber}";
            savedDetailsUrl = $"/cases/{savedCase.Id}";
            stage = ChatStage.Saved;
            formalReportConsentConfirmed = false;
        }
        catch
        {
            errorMessage = "تعذر حفظ الطلب. إجاباتك ما زالت هنا؛ حاول الحفظ مرة أخرى.";
        }
        finally { isSaving = false; }
    }
}
